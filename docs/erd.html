    USERS {
        bigint id PK
        string email UK
        string password_hash
        string full_name
        string avatar_url
        enum role "ROLE_USER, ROLE_ADMIN"
        enum status "ACTIVE, INACTIVE, BLOCKED"
        timestamp created_at
        timestamp updated_at
    }

    REFRESH_TOKENS {
        bigint id PK
        bigint user_id FK
        string token UK
        string device_info
        timestamp expires_at
        timestamp created_at
    }

    USER_SETTINGS {
        bigint id PK
        bigint user_id FK,UK
        int daily_new_cards_goal
        int daily_review_cards_goal
        time reminder_time
        boolean sound_enabled
        timestamp updated_at
    }

    DECKS {
        bigint id PK
        bigint user_id FK "Owner"
        bigint parent_deck_id FK "Nullable - for cloned decks"
        string title
        text description
        string cover_image_url
        string source_language "en"
        string target_language "vi"
        enum visibility "PRIVATE, PUBLIC"
        boolean is_official
        int total_cards
        int fork_count
        boolean is_deleted
        timestamp created_at
        timestamp updated_at
    }

    TAGS {
        bigint id PK
        string name UK "TOEIC, IELTS, IT English"
        string slug UK
    }

    DECK_TAGS {
        bigint deck_id PK,FK
        bigint tag_id PK,FK
    }

    CARDS {
        bigint id PK
        bigint deck_id FK
        string word
        string phonetic
        string part_of_speech "Noun, Verb, Adj..."
        text meaning
        text definition_en
        text example_sentence
        text example_meaning
        string image_url
        string audio_url
        text note
        int position
        timestamp created_at
        timestamp updated_at
    }

    USER_CARD_SRS {
        bigint id PK
        bigint user_id FK
        bigint card_id FK
        enum status "NEW, LEARNING, MASTERED"
        float easiness_factor "Default 2.5"
        int repetitions "Default 0"
        int interval "Default 0 (days)"
        timestamp last_reviewed_at
        date next_review_date "Indexed for Study Queue"
        timestamp created_at
        timestamp updated_at
    }

    QUIZ_RESULTS {
        bigint id PK
        bigint user_id FK
        enum scope "DECK, ALL, TAG"
        int total_questions
        int correct_answers
        float score
        float accuracy_rate
        int time_spent_seconds
        timestamp created_at
    }

    QUIZ_DETAILS {
        bigint id PK
        bigint quiz_result_id FK
        bigint card_id FK
        enum question_type "MULTIPLE_CHOICE, FILL_IN_BLANK, MATCHING, LISTENING"
        text user_answer
        text correct_answer
        boolean is_correct
        float score_earned
    }

    USER_ACTIVITIES {
        bigint id PK
        bigint user_id FK
        date activity_date
        int cards_reviewed_count
        int quizzes_completed_count
        timestamp created_at
    }

    USER_STREAKS {
        bigint id PK
        bigint user_id FK,UK
        int current_streak
        int longest_streak
        date last_active_date
        timestamp updated_at
    }

    BATTLES {
        bigint id PK
        bigint player1_id FK "Host"
        bigint player2_id FK "Opponent"
        bigint deck_id FK "Battle Deck"
        enum status "WAITING, IN_PROGRESS, COMPLETED, CANCELLED"
        bigint winner_id FK "Nullable"
        float player1_score
        float player2_score
        int player1_time_seconds
        int player2_time_seconds
        timestamp created_at
        timestamp completed_at
    }

    BATTLE_DETAILS {
        bigint id PK
        bigint battle_id FK
        bigint card_id FK
        text player1_answer
        text player2_answer
        boolean player1_is_correct
        boolean player2_is_correct
        int player1_time_ms
        int player2_time_ms
    }

    BADGES {
        bigint id PK
        string code UK "STREAK_7_DAYS, BATTLE_WINNER_10"
        string name "Thánh Chăm Chỉ, Vua Đấu Trí"
        text description
        string icon_url
        int points "Bonus Points"
    }

    USER_BADGES {
        bigint id PK
        bigint user_id FK
        bigint badge_id FK
        timestamp unlocked_at
    }

    LEADERBOARD_SNAPSHOTS {
        bigint id PK
        bigint user_id FK
        enum period_type "WEEKLY, MONTHLY, ALL_TIME"
        string period_key "2026-W32, 2026-08, GLOBAL"
        int total_points
        int rank_position
        timestamp updated_at
    }

    NOTIFICATIONS {
        bigint id PK
        bigint user_id FK
        enum type "BATTLE_INVITE, STREAK_REMINDER, BADGE_UNLOCKED, SYSTEM"
        string title
        text message
        string target_url
        boolean is_read "Default false"
        timestamp created_at
    }
